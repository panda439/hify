package provider

import (
	"context"
	"testing"
	"time"
)

func TestReviewP3TimeoutIncludesConcurrencyWait(t *testing.T) {
	c := WithResilience(&countingClient{}, ResilienceConfig{ProviderID: "review-timeout", MaxConcurrent: 1}).(*resilientClient)
	if err := c.sem.Acquire(t.Context(), 1); err != nil {
		t.Fatal(err)
	}
	defer c.sem.Release(1)
	ctx, cancel := context.WithTimeout(t.Context(), 200*time.Millisecond)
	defer cancel()
	start := time.Now()
	_, err := c.ChatOnce(ctx, ChatRequest{}, 20*time.Millisecond)
	if err != nil {
		t.Fatal(err)
	}
	if elapsed := time.Since(start); elapsed > 100*time.Millisecond {
		t.Fatalf("20ms request timeout did not bound concurrency wait: %s", elapsed)
	}
}
